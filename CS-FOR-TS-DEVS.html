3. Namespace = last-name NOT file/folder

// NavNet/Dtos/GameDto.cs:4
namespace NavNet.Dtos;
public record class GameDto(...);  → full name = NavNet.Dtos.GameDto

// NavNet/Models/Game.cs:1
namespace NavNet.Models;
public class Game {...}  → NavNet.Models.Game
using NavNet.Dtos;
var x = new GameDto(...); // short
var y = new NavNet.Dtos.GameDto(...);
// long, no using needed
⚠ Bad org that still works (verified with dotnet run):
// File: Demo/WrongFolder/FileA.cs — folder≠file≠namespace, still compiles
namespace Demo.Dtos;
public class Bar { ... }
Compiler looks at namespace, never at path.
